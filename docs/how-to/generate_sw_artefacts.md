# How-to: Generate SW Artefacts (C Header and Python Driver)

Zuspec can generate two software artefacts directly from a `RegisterFile`
declaration:

- A **C header** with `#define` macros for register offsets, field
  shifts/masks, and MMIO access macros.
- A **Python driver** class with typed field-level accessors that wrap a
  user-supplied `read`/`write` callable.

Both generators live in `zuspec-be-sw`.

---

## Prerequisites

```bash
pip install zuspec-be-sw   # or install from packages/zuspec-be-sw
```

---

## Example register file

```python
import zuspec.dataclasses as zdc

@zdc.regfile
class DMARegs(zdc.RegisterFile):

    @zdc.reg(offset=0x00)
    class CTRL:
        START: zdc.u1 = zdc.FieldAttr.Pulse
        ABORT: zdc.u1 = zdc.FieldAttr.Pulse

    @zdc.reg(offset=0x04)
    class STATUS:
        BUSY:  zdc.u1 = zdc.reg_field(sw=zdc.SW.RO, hw=zdc.HW.W,
                                       hwset=True, hwclr=True, default=0)
        DONE:  zdc.u1 = zdc.FieldAttr.StickyBit
        ERROR: zdc.u1 = zdc.FieldAttr.StickyBit
```

---

## C header generation

```python
from zuspec.be.sw.mmr import emit_c_header

h_text = emit_c_header(DMARegs)
with open("dma_regs.h", "w") as f:
    f.write(h_text)
```

### Generated output (excerpt)

```c
/* Auto-generated by zuspec MmrRegFileCHeaderEmitter */
/* Source: DMARegs */
#ifndef DMA_REGS_H
#define DMA_REGS_H

#include <stdint.h>

/* Register offsets */
#define DMA_REGS_CTRL_OFFSET                         0x00U
#define DMA_REGS_STATUS_OFFSET                       0x04U

/* CTRL fields */
#define DMA_REGS_CTRL_START_SHIFT                    0U
#define DMA_REGS_CTRL_START_MASK                     (0x1U << DMA_REGS_CTRL_START_SHIFT)
#define DMA_REGS_CTRL_ABORT_SHIFT                    1U
#define DMA_REGS_CTRL_ABORT_MASK                     (0x1U << DMA_REGS_CTRL_ABORT_SHIFT)

/* STATUS fields */
#define DMA_REGS_STATUS_BUSY_SHIFT                   0U
#define DMA_REGS_STATUS_BUSY_MASK                    (0x1U << DMA_REGS_STATUS_BUSY_SHIFT)
#define DMA_REGS_STATUS_DONE_SHIFT                   1U
#define DMA_REGS_STATUS_DONE_MASK                    (0x1U << DMA_REGS_STATUS_DONE_SHIFT)

/* Generic MMIO access */
#define DMA_REGS_RD(base, off)        (*(volatile uint32_t *)((uintptr_t)(base) + (off)))
#define DMA_REGS_WR(base, off, val)   (*(volatile uint32_t *)((uintptr_t)(base) + (off)) = (uint32_t)(val))
#define DMA_REGS_RMW(base, off, mask, shift, val) \
    DMA_REGS_WR((base), (off), (DMA_REGS_RD((base), (off)) & ~(mask)) | \
        (((uint32_t)(val) << (shift)) & (mask)))

/* Field access macros */
#define DMA_REGS_CTRL_START_RD(base)                 \
    ((uint32_t)(DMA_REGS_RD((base), DMA_REGS_CTRL_OFFSET) & DMA_REGS_CTRL_START_MASK) >> DMA_REGS_CTRL_START_SHIFT)
#define DMA_REGS_CTRL_START_WR(base, val)            \
    DMA_REGS_RMW((base), DMA_REGS_CTRL_OFFSET, DMA_REGS_CTRL_START_MASK, DMA_REGS_CTRL_START_SHIFT, (val))

#define DMA_REGS_STATUS_BUSY_RD(base)                \
    ((uint32_t)(DMA_REGS_RD((base), DMA_REGS_STATUS_OFFSET) & DMA_REGS_STATUS_BUSY_MASK) >> DMA_REGS_STATUS_BUSY_SHIFT)
/* DMA_REGS_STATUS_BUSY_WR omitted — SW.RO field */

#define DMA_REGS_STATUS_DONE_RD(base)                \
    ((uint32_t)(DMA_REGS_RD((base), DMA_REGS_STATUS_OFFSET) & DMA_REGS_STATUS_DONE_MASK) >> DMA_REGS_STATUS_DONE_SHIFT)
#define DMA_REGS_STATUS_DONE_CLR(base)               \
    DMA_REGS_WR((base), DMA_REGS_STATUS_OFFSET, DMA_REGS_STATUS_DONE_MASK)

#endif /* DMA_REGS_H */
```

### Key behaviours

| Field property | Macro generated |
|---|---|
| `sw=SW.RW` | `_RD(base)` **and** `_WR(base, val)` |
| `sw=SW.RO` | `_RD(base)` only — no `_WR` |
| `sw=SW.WO` | `_WR(base, val)` only — no `_RD` |
| `onwrite='woclr'` | `_CLR(base)` instead of `_WR(base, val)` |

### Options

```python
emit_c_header(
    DMARegs,
    prefix="DRV",          # override macro prefix (default: UPPER_SNAKE class name)
    data_type="uint16_t",  # override register data type (default: uint32_t)
    include_stdint=False,  # omit #include <stdint.h>
)
```

## Python driver generation

```python
from zuspec.be.sw.mmr import emit_py_driver

py_text = emit_py_driver(DMARegs)
with open("dma_regs_driver.py", "w") as f:
    f.write(py_text)
```

### Using the generated driver

```python
# Generated class — DMARegsDriver
from dma_regs_driver import DMARegsDriver

# Wrap your platform's MMIO read/write
import mmap, os

fd   = os.open("/dev/mem", os.O_RDWR | os.O_SYNC)
mem  = mmap.mmap(fd, 0x1000, offset=0x4000_0000)

def rd(addr):       return int.from_bytes(mem[addr:addr+4], "little")
def wr(addr, val):  mem[addr:addr+4] = val.to_bytes(4, "little")

regs = DMARegsDriver(base_addr=0, read_fn=rd, write_fn=wr)

# Kick off a DMA transfer
regs.set_CTRL_START(1)

# Poll until done
import time
while regs.get_STATUS_DONE() == 0:
    time.sleep(0.001)

# Clear the stickybit
regs.clr_STATUS_DONE()
```

### Generated methods

| Field property | Methods generated |
|---|---|
| `sw=SW.RW`, no `woclr` | `get_REG_FIELD()` **and** `set_REG_FIELD(value)` |
| `sw=SW.RO` | `get_REG_FIELD()` only |
| `sw=SW.WO` | `set_REG_FIELD(value)` only |
| `onwrite='woclr'` | `get_REG_FIELD()` **and** `clr_REG_FIELD()` (no `set_`) |

Every register also gets `read_REG()` and `write_REG(value)` for raw 32-bit access.

### Custom class name

```python
emit_py_driver(DMARegs, class_name="DMARegisterDriver")
```

---

## Both artefacts together

```python
from zuspec.be.sw.mmr import emit_c_header, emit_py_driver

with open("dma_regs.h",           "w") as f: f.write(emit_c_header(DMARegs))
with open("dma_regs_driver.py",   "w") as f: f.write(emit_py_driver(DMARegs))
```
